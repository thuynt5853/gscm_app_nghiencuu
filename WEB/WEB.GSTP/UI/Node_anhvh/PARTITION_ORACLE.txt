select tt.TOAANID,tt.emp_id from (
select CB.TOAANID,MAX(CB.ID) OVER (PARTITION BY CB.TOAANID ORDER BY CB.ID DESC ) AS emp_id  
    from dm_canbo cb 
    )tt
     WHERE tt.TOAANID IN(29,27)
    group by tt.TOAANID,tt.emp_id;
